from datetime import date

from dips_audit.discrepancias import Discrepancias
from dips_audit.reglas import Contexto, auditar_dip, etapa
from tests.conftest import hacer_dip


def contexto(activa=True):
    configuracion = Discrepancias.desde_dict({
        'Datos requeridos': {"reglas": {
            'descripcion_obligatoria': {"activa": activa, "severidad": 'OBSERVACION'}
        }}
    })
    return Contexto(fecha_referencia=date(2026, 10, 5), discrepancias=configuracion)


def codigos(dip, activa=True):
    return {h.codigo for h in auditar_dip(dip, None, contexto(activa)).hallazgos}


def test_descripcion_obligatoria():
    falla = hacer_dip(estado='ACTV', descripcion='')
    cumple = {**falla, 'descripcion': 'Dato presente'}
    assert 'descripcion_obligatoria' in codigos(falla)
    assert 'descripcion_obligatoria' not in codigos(cumple)
    assert 'descripcion_obligatoria' not in codigos(falla, activa=False)
    otra_etapa = {**falla, "estado": 'COMPLETE'}
    assert 'descripcion_obligatoria' not in codigos(otra_etapa)
    assert etapa('descripcion_obligatoria') == 'APERTURA'
